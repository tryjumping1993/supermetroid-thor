cmake_minimum_required(VERSION 3.22)
project(supermetroid_thor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(THOR_BUILD_TESTS "Build portable native verification executable" ON)
file(GLOB_RECURSE THOR_CORE_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(FILTER THOR_CORE_SOURCES EXCLUDE REGEX "src/(android_bridge|renderer)\.cpp$")
add_library(thor_core STATIC ${THOR_CORE_SOURCES})
set_target_properties(thor_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(thor_core PUBLIC include generated)
target_compile_options(thor_core PRIVATE -Wall -Wextra -Wpedantic)
if(ANDROID)
    add_library(thor SHARED src/android_bridge.cpp src/renderer.cpp)
    target_link_libraries(thor PRIVATE thor_core GLESv3 android log)
    target_compile_options(thor PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(THOR_BUILD_TESTS)
    add_executable(thor_tests tests/core_tests.cpp)
    target_link_libraries(thor_tests PRIVATE thor_core)
    enable_testing()
    add_test(NAME thor_core COMMAND thor_tests)
endif()
option(THOR_BUILD_TOOLS "Build the route recorder (tools/route_probe)" OFF)
if(THOR_BUILD_TOOLS)
    add_executable(route_probe ../tools/route_probe/route_probe.cpp)
    target_link_libraries(route_probe PRIVATE thor_core)
endif()
